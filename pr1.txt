import java.util.Scanner;

public class Fibonacci {

    // Non-recursive method
    static void iterativeFibonacci(int n) {
        int a = 0, b = 1, c;

        System.out.println("Fibonacci using Iterative Method:");

        for (int i = 0; i < n; i++) {
            System.out.print(a + " ");

            c = a + b;
            a = b;
            b = c;
        }

        System.out.println();
    }

    // Recursive method
    static int recursiveFibonacci(int n) {
        if (n == 0)
            return 0;

        if (n == 1)
            return 1;

        return recursiveFibonacci(n - 1)
             + recursiveFibonacci(n - 2);
    }

    public static void main(String[] args) {

        Scanner sc = new Scanner(System.in);

        System.out.print("Enter number of terms: ");
        int n = sc.nextInt();

        if (n <= 0) {
            System.out.println("Please enter a positive number.");
        } else {

            // Iterative Fibonacci
            iterativeFibonacci(n);

            // Recursive Fibonacci
            System.out.println("Fibonacci using Recursive Method:");

            for (int i = 0; i < n; i++) {
                System.out.print(recursiveFibonacci(i) + " ");
            }

            System.out.println();
        }

        sc.close();
    }
}

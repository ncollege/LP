import java.util.*;

class Node {
    char ch;
    int frequency;

    Node left, right;

    Node(char ch, int frequency) {
        this.ch = ch;
        this.frequency = frequency;
        left = right = null;
    }
}

public class HuffmanEncoding {

    // Generate Huffman codes
    static void generateCodes(Node root, String code) {

        if (root == null)
            return;

        // If leaf node, print character and its code
        if (root.left == null && root.right == null) {
            System.out.println(root.ch + " : " + code);
            return;
        }

        generateCodes(root.left, code + "0");
        generateCodes(root.right, code + "1");
    }

    public static void main(String[] args) {

        Scanner sc = new Scanner(System.in);

        System.out.print("Enter number of characters: ");
        int n = sc.nextInt();

        char[] characters = new char[n];
        int[] frequencies = new int[n];

        // Input characters and frequencies
        for (int i = 0; i < n; i++) {

            System.out.print("Enter character: ");
            characters[i] = sc.next().charAt(0);

            System.out.print("Enter frequency: ");
            frequencies[i] = sc.nextInt();
        }

        // Create priority queue
        PriorityQueue<Node> pq = new PriorityQueue<>(
            (a, b) -> Integer.compare(a.frequency, b.frequency)
        );

        // Insert nodes into priority queue
        for (int i = 0; i < n; i++) {
            pq.add(new Node(characters[i], frequencies[i]));
        }

        // Build Huffman tree
        while (pq.size() > 1) {

            Node left = pq.poll();
            Node right = pq.poll();

            Node newNode = new Node(
                '-', left.frequency + right.frequency
            );

            newNode.left = left;
            newNode.right = right;

            pq.add(newNode);
        }

        // Root of Huffman tree
        Node root = pq.poll();

        // Display Huffman codes
        System.out.println("\nHuffman Codes:");

        if (n == 1) {
            System.out.println(characters[0] + " : 0");
        } else {
            generateCodes(root, "");
        }

        sc.close();
    }
}

pragma solidity ^0.8.20;

/// @title Customer Bank Account
/// @notice A simple test-network smart contract supporting deposit, withdrawal, and balance checking.
contract CustomerBankAccount {
    address public immutable customer;

    event Deposited(address indexed from, uint256 amount);
    event Withdrawn(address indexed to, uint256 amount);

    constructor() {
        customer = msg.sender;
    }

    modifier onlyCustomer() {
        require(msg.sender == customer, "Only the customer can perform this action");
        _;
    }

    // Deposit ETH into the customer's account.
    function deposit() external payable onlyCustomer {
        require(msg.value > 0, "Deposit amount must be greater than zero");
        emit Deposited(msg.sender, msg.value);
    }

    // Withdraw ETH from the customer's account.
    function withdraw(uint256 amount) external onlyCustomer {
        require(amount > 0, "Withdrawal amount must be greater than zero");
        require(amount <= address(this).balance, "Insufficient balance");

        (bool success, ) = payable(customer).call{value: amount}("");
        require(success, "Withdrawal failed");

        emit Withdrawn(customer, amount);
    }

    // Show the current contract balance in wei.
    function showBalance() external view returns (uint256) {
        return address(this).balance;
    }
}
